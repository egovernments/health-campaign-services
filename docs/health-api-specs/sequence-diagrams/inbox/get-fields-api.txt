@startuml
title Inbox - V2 Get Fields (flattened ES index search)
!theme vibrant
participant Client as c
participant InboxService as s
participant MDMSService as mdms
participant ElasticSearch as el

c -> s : POST /inbox/v2/_getFields
activate s

s -> s: Validate request body (@Valid SearchRequest)
note right
    SearchCriteria is not annotated @Valid, so
    tenantId / moduleName @NotNull and limit @Max(300)
    are NOT enforced. Missing SearchCriteria or
    moduleSearchCriteria -> 400 NullPointerException.
end note
alt request body cannot be parsed
    s -> c: HttpStatus: 400 { Errors }
end

group validateSearchCriteria(tenantId, moduleName, moduleSearchCriteria)
    s -> mdms: /egov-mdms-service/v1/_search\n(tenantId = state level tenant, module INBOX,\nmaster InboxQueryConfiguration)
    activate mdms
    note right
        @Cacheable "inboxConfiguration" (cache2k, TTL 10 min).
        Filter [?(@.module == SearchCriteria.moduleName)]
    end note
    alt MDMS call / JsonPath fails
        mdms -> s: Error
        s -> c: HttpStatus: 400 CONFIG_ERROR\n(Error in fetching inbox query configuration from MDMS)
    else no config for moduleName
        mdms -> s: Empty
        s -> c: HttpStatus: 400 CONFIG_ERROR\n(Inbox Query Configuration not found in MDMS response)
    else config found
        mdms -> s: InboxQueryConfiguration
        deactivate mdms
    end
    loop each moduleSearchCriteria key (except sortBy / sortOrder)
        alt key not present in allowedSearchCriteria
            s -> c: HttpStatus: 400 NullPointerException
        else mandatory key with null / empty value
            s -> c: HttpStatus: 400 INVALID_SEARCH_CRITERIA\n(Field cannot be null or empty: <key>)
        end
    end
    note right
        Unlike /v2/_search, missing mandatory keys
        are not checked here.
    end note
end

s -> mdms: getConfigFromMDMS(tenantId, moduleName)
activate mdms
mdms -> s: InboxQueryConfiguration (served from cache)
deactivate mdms

s -> s: Hash values of params with isHashingRequired = true\n(SHA-256, local)

s -> s: Build ES query: from = offset, size = limit,\nsort [ sortBy.path : sortOrder or defaultOrder ],\n_source = sourceFilterPathList,\nbool.must = moduleSearchCriteria clauses only
note right
    No tenantId / status / role based filtering
    is added for this endpoint.
end note
s -> el: POST <services.esindexer.host><index>/_search\n(Authorization: Basic egov.es.username:password)
activate el
alt ES call fails
    el -> s: Error
    s -> c: HttpStatus: 400 { ES error body }
else
    el -> s: hits
    deactivate el
end

loop each hit
    s -> s: Flatten hit._source.Data (JsonFlattener)\ninto fields [ { key, value } ]
    alt JSON processing fails
        s -> c: HttpStatus: 400 EG_INBOX_GET_FIELDS_ERR
    end
end

s -> c: HttpStatus: 200 { data: [ { fields: [ { key, value } ] } ] }
deactivate s

@enduml
