@startuml
title Resource Generator - Create Draft Microplan (Drafts)
!theme vibrant
participant Client as c
participant ResourceGenerator as s
participant PlanService as ps
participant FileStore as fs
participant ProjectFactory as pf
participant LocalizationService as loc
participant MDMSService as mdms
participant BoundaryService as bs
queue Kafka as k
participant HandleErrorMethod as he

c -> s : POST /resource-generator/drafts
activate s

s -> s: Validate request body (@Valid)\n(RequestInfo not null; DraftDetails not null;\nplanConfigurationId, tenantId not null, size 2..64)
alt request body validation fails
    s -> c: HttpStatus: 400 { Errors }
end

s -> s: draftService.createDraftPlans(draftRequest) (@Async)
s -> c: HttpStatus: 202 "Draft processing started successfully."
deactivate s

group async createDraftPlans
    s -> s: Run on @Async executor thread
    activate s
    s -> ps: /plan-service/config/_search\n(id = planConfigurationId, tenantId)
    activate ps
    ps -> s: PlanConfigurationResponse
    deactivate ps
    note right
        Search errors are only logged; an empty result
        fails on planConfigurationList.get(0)
    end note

    s -> s: fileStoreId = active file with\ntemplateIdentifier "Population"
    s -> fs: GET /filestore/v1/files/id?tenantId={tenantId}&fileStoreId={fileStoreId}
    activate fs
    fs -> s: File bytes
    deactivate fs
    alt download fails
        s -> s: CustomException FILESTORE_EXCEPTION
    else file missing on disk
        s -> s: CustomException FILE_NOT_FOUND
    else workbook cannot be read
        s -> s: CustomException WORKBOOK_READ_ERROR
    end

    s -> pf: /project-factory/v1/project-type/search\n(CampaignDetails ids = [campaignId], tenantId)
    activate pf
    pf -> s: CampaignResponse
    deactivate pf
    alt campaign search fails
        s -> s: CustomException ERROR_FETCHING_CAMPAIGN_DETAILS
    end

    s -> loc: /localization/messages/v1/_search?module=rainmaker-microplanning,\nrainmaker-boundary-undefined,hcm-admin-schemas&locale=<msgId locale>&tenantId
    activate loc
    loc -> s: LocaleResponse
    deactivate loc

    s -> mdms: /egov-mdms-service/v1/_search\n(HCM-ADMIN-CONSOLE.adminSchema)
    activate mdms
    mdms -> s: MDMS data
    deactivate mdms
    alt MDMS response empty
        s -> s: CustomException NO_MDMS_DATA_FOUND_FOR_GIVEN_TENANT
    end

    s -> s: Sort operations by executionOrder

    group enrichResourceMapping
        s -> mdms: /mdms-v2/v2/_search (root tenant,\nschemaCode HCM-ADMIN-CONSOLE.adminSchema,\nuniqueIdentifier boundary.MP-<projectType>)
        activate mdms
        mdms -> s: Admin schema (number / string properties)
        deactivate mdms
        s -> s: Deactivate existing resourceMapping,\nadd one active mapping per schema column\n(mappedFrom = localised column name)
    end

    s -> s: Filter adminSchema properties for\ntitle boundaryWithTarget, campaignType MP-<projectType>

    s -> bs: /boundary-service/boundary-relationships/_search\n?includeChildren=true&tenantId&hierarchyType
    activate bs
    bs -> s: BoundarySearchResponse
    deactivate bs
    alt boundary search fails
        s -> s: CustomException BOUNDARY_SEARCH_EXCEPTION
    else no tenantBoundary
        s -> s: CustomException BOUNDARY_CODE_NOT_FOUND
    end

    s -> mdms: /egov-mdms-service/v1/_search\n(hcm-microplanning.CommonConstants)
    activate mdms
    mdms -> s: Common constants (readMeSheetName)
    deactivate mdms

    loop each sheet except README / HCM_ADMIN_CONSOLE_BOUNDARY_DATA sheet
        s -> mdms: /mdms-v2/v2/_search (root tenant,\nhcm-microplanning.MixedStrategyOperationLogic)
        activate mdms
        mdms -> s: Mixed strategy logic (fetched but unused for drafts)
        deactivate mdms
        alt MDMS v2 response empty
            s -> s: CustomException NO_MDMS_DATA_FOUND_FOR_GIVEN_TENANT
        end
        loop each non-empty data row
            s -> s: validateRows (cells before boundary code,\nboundary code exists, required / numeric / boolean cells)
            alt row invalid
                s -> he: catch in validateRows
                activate he
                he -> he: PlanConfiguration.status = INVALID_DATA
                he -> k: PlanConfigurationRequest /resource-plan-config-update-topic
                activate k
                he -> s: CustomException 500 "File does not contain valid input for row ..."
                deactivate he
            end
            s -> s: Calculate each operation, write outputs into new cells\n(skipFixedPost = true, no mixed strategy)
            opt PlanConfiguration.status != DRAFT
                s -> k: PlanRequest (workflow INITIATE)\n/resource-microplan-create-topic
                activate k
            end
        end
    end

    group processDraftOutputFile
        s -> loc: /localization/messages/v1/_search
        activate loc
        loc -> s: LocaleResponse
        deactivate loc
        s -> mdms: /egov-mdms-service/v1/_search (CommonConstants)
        activate mdms
        mdms -> s: Common constants
        deactivate mdms
        s -> s: Remove README / boundary data sheets,\nlocalise header row, append additionalDetails.newColumns
        alt header row empty
            s -> s: CustomException EMPTY_HEADER_ROW
        end
    end

    s -> fs: POST /filestore/v1/files (multipart, tenantId, module = microplan)
    activate fs
    fs -> s: files[0].fileStoreId
    deactivate fs

    s -> s: Mark existing "DraftComplete" file inactive,\nadd new "DraftComplete" EXCEL file, workflow = null
    s -> k: PlanConfigurationRequest /resource-plan-config-update-topic
    activate k
    deactivate s
    note right
        Any exception in this async block is only
        visible in logs; the client already got 202
    end note
end

group async downstream
    ps -> k: Consume /resource-plan-config-update-topic
    activate ps
    ps -> ps: PlanConfigurationService.update(...)
    ps -> k: Consume /resource-microplan-create-topic
    ps -> ps: PlanService.createPlan(...)
    deactivate ps
end
deactivate k

@enduml
