@startuml
title Notification Push - Send Push Notification
!theme vibrant
participant Client as c
participant NotificationPushService as s
participant FirebaseCloudMessaging as fcm
database Database as db

c -> s : POST /egov-notification-push/push/v1/_send
activate s

s -> s: Validate request body (@Valid)\n(RequestInfo @NotNull, tenantId @NotNull)
alt request body validation fails
    s -> c: HttpStatus: 400 { Errors }
end

group validateRequest
    alt title is blank
        s -> c: HttpStatus: 400 PUSH_MISSING_TITLE
    else body is blank
        s -> c: HttpStatus: 400 PUSH_MISSING_BODY
    else userUuids and deviceTokens both empty
        s -> c: HttpStatus: 400 PUSH_MISSING_RECIPIENTS
    end
end

opt userUuids present
    s -> s: Resolve schema from tenantId
    alt central instance and tenantId invalid
        s -> c: HttpStatus: 400 MISSING_TENANT_ID / INVALID_TENANT_ID
    end
    s -> db: SELECT ... FROM <schema>.eg_push_device_tokens\nWHERE userid IN (:userUuids)
    activate db
    db -> s: Device token rows (all facilities)
    deactivate db
    s -> s: Add device tokens to ordered set
end
opt deviceTokens present
    s -> s: Add request deviceTokens to set (de-duplicated)
end

alt no tokens resolved
    s -> c: HttpStatus: 200 "Push notification sent to 0 devices"
else tokens resolved
    s -> s: Build PushNotificationRequest\n(title, body, data, deviceTokens, tenantId)
    alt fcm.enabled = true (FirebasePushService)
        alt exactly one token
            s -> fcm: FirebaseMessaging.send(Message)
            activate fcm
            fcm -> s: messageId / FirebaseMessagingException
            deactivate fcm
        else more than one token
            loop batches of fcm.batch.size (500)
                s -> fcm: FirebaseMessaging.sendEachForMulticast(MulticastMessage)
                activate fcm
                fcm -> s: BatchResponse (success / failure per token)
                deactivate fcm
            end
        end
        opt token failed with UNREGISTERED
            s -> db: DELETE FROM <schema>.eg_push_device_tokens\nWHERE devicetoken IN (:deviceTokens)
            activate db
            deactivate db
        end
    else fcm.enabled = false (ConsolePushService)
        s -> s: Log title, body, data, device tokens
    end
    s -> c: HttpStatus: 200 "Push notification sent to <n> devices"
end
deactivate s

note over s, fcm
    FCM errors are caught and logged; the API still returns 200 and
    <n> is the number of targeted tokens, not successful deliveries.
    Stale-token delete is a direct JDBC delete (not via Kafka/persister)
    and removes the token for all users and facilities.
end note

@enduml
