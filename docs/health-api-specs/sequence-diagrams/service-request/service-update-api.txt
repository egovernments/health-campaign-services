@startuml
title Service Request - Service Update
!theme vibrant
participant Client as c
participant ServiceRequestService as s
queue Kafka as k
participant PersisterService as prs
database Database as db

c -> s : POST /service-request/service/v1/_update
activate s

s -> s: Validate request body (@Valid)\n(same rules as create: tenantId, serviceDefId,\nattributes, accountId required)
alt request body validation fails
    s -> c: HttpStatus: 400 { Errors }
end

group validateServiceUpdateRequest
    group validateExistingServiceRequest
        s -> db: SELECT eg_service INNER JOIN eg_service_attribute_value\nWHERE service.tenantid = ? AND service.id IN (?)
        activate db
        db -> s: Existing service
        deactivate db
        alt service not found
            s -> c: HttpStatus: 400 SERVICE_NOT_EXIST_ERR_CODE
        end
    end

    group validateServiceDefID
        s -> db: SELECT eg_service_definition INNER JOIN\neg_service_attribute_definition WHERE sd.tenantid = ? AND sd.id IN (?)
        activate db
        db -> s: Service definition
        deactivate db
        alt definition not found
            s -> c: HttpStatus: 400 SERVICE_REQUEST_INVALID_SERVICE_DEF_ID
        end
    end

    group validateIdForAttributeValues
        alt duplicate attribute value id (request or DB)
            s -> c: HttpStatus: 400 SERVICE_ATTRIBUTE_VALUE_INVALID_ERR_CODE\n(Service attribute value id is duplicated)
        end
        alt attribute id not part of existing service
            s -> c: HttpStatus: 400 SERVICE_ATTRIBUTE_VALUE_INVALID_ERR_CODE\n(Service attribute value does not exist)
        end
        alt attribute without id whose attributeCode already exists
            s -> c: HttpStatus: 400 SERVICE_ATTRIBUTE_VALUE_INVALID_ERR_CODE\n(code already existing but no id is passed)
        end
        s -> s: Merge: existing attributes replaced by request attributes\nwith same id, new attributes (no id) appended
    end

    group validateAttributeValuesAgainstServiceDefinition (merged list)
        alt attributeCode not in definition
            s -> c: HttpStatus: 400 SERVICE_REQUEST_UNRECOGNIZED_ATTRIBUTE_CODE
        end
        alt same attributeCode twice
            s -> c: HttpStatus: 400 SERVICE_REQUEST_ATTRIBUTE_VALUES_UNIQUENESS_ERR_CODE
        end
        alt required attribute missing
            s -> c: HttpStatus: 400 SERVICE_REQUEST_REQUIRED_ATTRIBUTE_NOT_PROVIDED_ERR_CODE
        end
        alt value type does not match dataType or list value not allowed
            s -> c: HttpStatus: 400 SERVICE_REQUEST_ATTRIBUTE_INVALID_VALUE_CODE
        end
        alt String / Text longer than 8192
            s -> c: HttpStatus: 400 INVALID_SIZE_OF_INPUT_CODE
        end
    end
    s -> s: Service.attributes = merged list;\nvalidateAccountId (empty, TODO)
end

group enrichServiceRequestUpdate
    s -> s: auditDetails: keep createdBy / createdTime from DB,\nlastModifiedBy = userInfo.uuid, lastModifiedTime = now
    loop for each merged attribute
        s -> s: id = random UUID if empty;\nkeep DB additionalFields if none passed;\nserviceClientReferenceId = clientId (if present);\nreferenceId = service id; auditDetails = updated
    end
    s -> s: Wrap each value as { "value": <value> }
end

s -> k: ServiceRequest /update-service
activate k
s -> s: Restore attribute values to native form
s -> c: HttpStatus: 200 Services [Service with merged attributes]
deactivate s

group async
    prs -> k: Consume /update-service
    activate prs
    prs -> db: UPDATE eg_service SET lastmodifiedby, lastmodifiedtime,\nadditionaldetails, accountId WHERE id = ?
    activate db
    deactivate db
    prs -> db: INSERT INTO eg_service_attribute_value ...\nON CONFLICT (id) DO UPDATE SET lastmodifiedby, lastmodifiedtime,\nattributecode, value, additionaldetails
    activate db
    deactivate db
    deactivate prs
end
deactivate k

note over prs, db
    ON CONFLICT (id) now targets pk_eg_service_attribute_value
    (was uk_eg_attribute_value before V20260918120000), so the upsert is idempotent.
    serviceDefId, referenceId and clientId of eg_service are never updated.
    No egov-indexer config exists for this topic.
end note

@enduml
